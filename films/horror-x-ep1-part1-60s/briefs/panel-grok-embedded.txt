export const clamp=(x,a=0,b=1)=>Math.max(a,Math.min(b,x));
export const ease=x=>{x=clamp(x);return x*x*(3-2*x);};
const esc=s=>String(s).replaceAll('&','&amp;').replaceAll('<','&lt;').replaceAll('>','&gt;');
export const rect=(x,y,w,h,fill,extra='')=>`<rect x="${x}" y="${y}" width="${w}" height="${h}" fill="${fill}" ${extra}/>`;
export const path=(d,fill='none',extra='')=>`<path d="${d}" fill="${fill}" ${extra}/>`;
export const line=(d,color='#8FB2D4',width=2,opacity=1)=>path(d,'none',`stroke="${color}" stroke-width="${width}" opacity="${opacity}" stroke-linecap="round" stroke-linejoin="round"`);
export const text=(x,y,s,size=32,fill='#EAF4FF',extra='')=>`<text x="${x}" y="${y}" font-size="${size}" fill="${fill}" font-family="Kanit" ${extra}>${esc(s)}</text>`;
export const stripSvg=s=>s.replace(/^\s*<svg\b[^>]*>/,'').replace(/<\/svg>\s*$/,'');
export const interior=t=>renderInterior(t);
export const exterior=t=>stripSvg(renderExterior(t));

export function view({back,mid='',front=''},u,d,{cx=540,cy=960,z0=1,z1=1.06,dx=0,dy=0}={}){

===== timeline.js =====
export const DURATION=60,FPS=30,WIDTH=1080,HEIGHT=1920;
export function sceneFor(t,scenes){
 const v=Math.max(0,Math.min(60-1e-7,Number(t)||0));
 const s=scenes.find(x=>v>=x.start&&v<x.end);
 if(!s)throw new Error(`No scene at${v}`);
 return s;
}
export function validateTimeline(scenes){
 if(scenes.length!==12)throw new Error('Exactly12 scenes required');
 let next=0;
 for(const s of scenes){if(Math.abs(s.start-next)>1e-4||!(s.end>s.start))throw new Error(`Timeline gap/overlap at${s.id}`);next=s.end;}
 if(Math.abs(next-60)>1e-4)throw new Error(`Timeline ends at${next}`);
 return true;
}

from pathlib import Path
import json,re,subprocess
ROOT=Path(__file__).resolve().parent
def run(args):
    r=subprocess.run(args,capture_output=True,text=True,encoding='utf-8',errors='replace')
    if r.returncode:raise RuntimeError(r.stderr[-5000:])
    return r
if __name__=='__main__':
    base='loudnorm=I=-18:TP=-1.5:LRA=11'
    measure=run(['ffmpeg','-hide_banner','-i',str(ROOT/'soundtrack.wav'),'-af',base+':print_format=json','-f','null','-'])
    m=json.loads(re.findall(r'\{\s*"input_i"[\s\S]*?\}',measure.stderr)[-1])
    filt=base+':linear=true:print_format=json'+''.join(f':{k}={m[v]}' for k,v in [('measured_I','input_i'),('measured_TP','input_tp'),('measured_LRA','input_lra'),('measured_thresh','input_thresh'),('offset','target_offset')])
    out=ROOT/'horror-x-ep1-part1-60s.mp4'
    result=run(['ffmpeg','-hide_banner','-y','-i',str(ROOT/'out/picture.mp4'),'-i',str(ROOT/'soundtrack.wav'),'-map','0:v:0','-map','1:a:0','-c:v','copy','-af',filt,'-c:a','aac','-b:a','192k','-ar','48000','-t','60','-movflags','+faststart',str(out)])
    run(['ffmpeg','-hide_banner','-v','error','-i',str(out),'-f','null','-'])
    probe=json.loads(run(['ffprobe','-v','error','-show_streams','-show_format','-of','json',str(out)]).stdout)
    v=next(x for x in probe['streams'] if x['codec_type']=='video')
    assert(v['width'],v['height'])==(1080,1920)
    assert v['r_frame_rate']=='30/1' and int(v['nb_frames'])==1800
    assert abs(float(probe['format']['duration'])-60)<.05
    (ROOT/'qa/final-probe.json').write_text(json.dumps(probe,indent=2),encoding='utf-8')
    (ROOT/'qa/mux-loudness.txt').write_text(result.stderr,encoding='utf-8')
    print('Verified60s,1800frames,1080x1920@30fps; normalized-18LUFS')
